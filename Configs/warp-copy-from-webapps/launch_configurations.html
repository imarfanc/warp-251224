import shutil
import sys
from pathlib import Path
from rich.console import Console
from rich.panel import Panel
from rich.prompt import Confirm

console = Console()

def prompt_overwrite(message: str) -> bool:
    # heredoc feeds script via stdin — read prompts from the terminal instead
    if sys.stdin.isatty():
        return Confirm.ask(message)
    try:
        with open("/dev/tty", "r") as tty:
            return Confirm.ask(message, stream=tty)
    except OSError:
        console.print("[yellow]Non-interactive — overwriting destination.[/yellow]")
        return True

# Define the realm of ideas (source) and the realm of execution (destination)
src = Path("~/Developer/gh/web-apps/Configs/warp/launch_configurations").expanduser()
dst_base = Path("~/.warp").expanduser()

# If ~/.warp exists as a directory, nest the configurations inside it
if dst_base.is_dir() and src.name != dst_base.name:
    dst = dst_base / src.name
else:
    dst = dst_base

console.print(Panel(
    f"[bold cyan]Source (Ideal):[/bold cyan] {src}\n"
    f"[bold magenta]Destination (Real):[/bold magenta] {dst}",
    title="[bold]Configuration Transfer[/bold]",
    expand=False
))

if not src.exists():
    console.print("[bold red]Error:[/bold red] The source of truth does not exist. Cannot manifest what is not there.")
    sys.exit(1)

if dst.exists():
    overwrite = prompt_overwrite("[yellow]The destination already exists. Overwrite the current state?[/yellow]")
    if not overwrite:
        console.print("[bold blue]Aborted.[/bold blue] The current state remains untouched.")
        sys.exit(0)
    if dst.is_dir():
        shutil.rmtree(dst)

try:
    if src.is_dir():
        shutil.copytree(src, dst)
    else:
        dst.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(src, dst)
    console.print("[bold green]Success:[/bold green] The ideal has been manifested in the runtime.", style="green")
except Exception as e:
    console.print(f"[bold red]Failure:[/bold red] The transfer was interrupted: {e}")
    sys.exit(1)
